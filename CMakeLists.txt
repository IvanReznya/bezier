cmake_minimum_required(VERSION 3.16)
project(Delaunay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()
option(BUILD_GUI "Build Qt desktop application" ON)
include(CTest)
add_library(Delaunay_geometry STATIC
    src/lib/geometry/point.cpp
    src/lib/geometry/polygons.cpp
    src/lib/geometry/triangulation.cpp
)
target_include_directories(Delaunay_geometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src/lib)
if(MSVC)
    target_compile_options(Delaunay_geometry PRIVATE /W4 /utf-8)
else()
    target_compile_options(Delaunay_geometry PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_GUI)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Widgets)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Widgets)
    add_library(Delaunay_geometry_qt STATIC src/lib/geometry_qt/qt_converters.cpp)
    target_link_libraries(Delaunay_geometry_qt PUBLIC Delaunay_geometry Qt${QT_VERSION_MAJOR}::Core)
    add_executable(Delaunay src/apps/GeometryWorkbench/main.cpp)
    target_link_libraries(Delaunay PRIVATE Delaunay_geometry_qt Qt${QT_VERSION_MAJOR}::Widgets)
    set_target_properties(Delaunay PROPERTIES MACOSX_BUNDLE TRUE WIN32_EXECUTABLE TRUE
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
    if(MSVC)
        target_compile_options(Delaunay PRIVATE /utf-8)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(Delaunay_test_polygons tests/geometry/test_polygons.cpp)
    target_link_libraries(Delaunay_test_polygons PRIVATE Delaunay_geometry)
    add_test(NAME 06_delaunay.polygons COMMAND Delaunay_test_polygons)
    add_executable(Delaunay_test_triangulation tests/geometry/test_triangulation.cpp)
    target_link_libraries(Delaunay_test_triangulation PRIVATE Delaunay_geometry)
    add_test(NAME 06_delaunay.triangulation COMMAND Delaunay_test_triangulation)
    if(BUILD_GUI)
        add_test(NAME 06_delaunay.gui COMMAND Delaunay --smoke-test)
        set_tests_properties(06_delaunay.gui PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 30)
    endif()
endif()
