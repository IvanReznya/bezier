cmake_minimum_required(VERSION 3.16)
project(BezierCurves LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()
option(BUILD_GUI "Build Qt desktop application" ON)
include(CTest)
add_library(BezierCurves_geometry STATIC
    src/lib/geometry/point.cpp
    src/lib/geometry/bezier.cpp
)
target_include_directories(BezierCurves_geometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src/lib)
if(MSVC)
    target_compile_options(BezierCurves_geometry PRIVATE /W4 /utf-8)
else()
    target_compile_options(BezierCurves_geometry PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_GUI)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core Widgets)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Widgets)
    add_library(BezierCurves_geometry_qt STATIC src/lib/geometry_qt/qt_converters.cpp)
    target_link_libraries(BezierCurves_geometry_qt PUBLIC BezierCurves_geometry Qt${QT_VERSION_MAJOR}::Core)
    add_executable(BezierCurves src/apps/GeometryWorkbench/main.cpp)
    target_link_libraries(BezierCurves PRIVATE BezierCurves_geometry_qt Qt${QT_VERSION_MAJOR}::Widgets)
    set_target_properties(BezierCurves PROPERTIES MACOSX_BUNDLE TRUE WIN32_EXECUTABLE TRUE
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
    if(MSVC)
        target_compile_options(BezierCurves PRIVATE /utf-8)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(BezierCurves_test_bezier tests/geometry/test_bezier.cpp)
    target_link_libraries(BezierCurves_test_bezier PRIVATE BezierCurves_geometry)
    add_test(NAME 13_bezier.bezier COMMAND BezierCurves_test_bezier)
    if(BUILD_GUI)
        add_test(NAME 13_bezier.gui COMMAND BezierCurves --smoke-test)
        set_tests_properties(13_bezier.gui PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 30)
    endif()
endif()
